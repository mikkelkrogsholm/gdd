class UniqueLoader(yaml.SafeLoader):pass
def mapping(loader,node,deep=False):
    out={}
    for kn,vn in node.value:
        k=loader.construct_object(kn,deep=deep)
        if k in out:raise ValueError('duplicate YAML key')
        out[k]=loader.construct_object(vn,deep=deep)
    return out
UniqueLoader.add_constructor(yaml.resolver.BaseResolver.DEFAULT_MAPPING_TAG,mapping)
def parse_yaml(text):
    for tok in yaml.scan(text):
        if isinstance(tok,(yaml.AnchorToken,yaml.AliasToken,yaml.TagToken)):raise ValueError('forbidden YAML feature')
    return yaml.load(text,Loader=UniqueLoader)

def patch(doc,ps):
    doc=copy.deepcopy(doc)
    for p in ps:
        parts=[x.replace('~1','/').replace('~0','~') for x in p['path'].split('/')[1:]];cur=doc
        for k in parts[:-1]:cur=cur[int(k)] if isinstance(cur,list) else cur[k]
        k=parts[-1];op=p['op']
        if isinstance(cur,list):
            if op=='add':cur.insert(len(cur) if k=='-' else int(k),copy.deepcopy(p['value']))
            elif op=='replace':cur[int(k)]=copy.deepcopy(p['value'])
            else:del cur[int(k)]
        elif op=='remove':del cur[k]
        else:cur[k]=copy.deepcopy(p['value'])
    return doc
